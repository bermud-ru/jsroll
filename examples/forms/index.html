jsroll ← ко всем примерам

Формы — group, валидация, typeahead, маски

Форма ниже собрана по мотивам карточки пользователя из модуля users (см. public/js/app/users.js + Application/view/app/users.phtml в загруженном проекте): та же связка group() + pattern_validator + maskedigits() + typeahead(), только без реального бэкенда — запросы typeahead и отправка формы уходят на пару «заглушек» прямо в этом файле (см. раздел про перехват XHR внизу страницы), чтобы пример был самодостаточным.

Формы jsroll — не только GET/POST. Нативный HTML5 <form method="…"> понимает только get/post (браузер молча подставит GET для любого другого значения). Но group/xhr() формы никогда не сабмитят её нативно — событие submit перехватывается (preventDefault), а сама отправка идёт через XMLHttpRequest, которому браузер разрешает открыть запрос ЛЮБЫМ методом. Поэтому атрибут method="put" (как в форме пользователя в оригинале: method="{%= user_id ? "put" : "post" %}") — это не опечатка и не «неверный HTML», а рабочий способ отправить полноценный REST-запрос PUT/PATCH/DELETE прямо из формы. Отдельная живая демонстрация — раздел 3 ниже.
1. Карточка пользователя — new group(formEl, opt)
Заполнение из JSON (кнопка «Заполнить из JSON» ниже) и чтение в JSON для отправки (кнопка «Показать JSON для отправки», раздел 5) устроены НЕ симметрично: чтение (form.data-геттер) само собирает name="address[city]"/name="address[zip]" во вложенный объект {address:{city,zip}}, а вот запись (form.data = {...}-сеттер) такой сборки не делает — ключи объекта должны буквально совпадать со строкой атрибута name, то есть {'address[city]': '…'}, а не {address:{city:'…'}}. Учитывайте это при заполнении форм данными с сервера. 


            
var form = new group(document.querySelector('#user-form'), Object.merge(group_xhr_opt, {
    // method здесь не указываем — group САМА читает form.getAttribute('method')
    done: function (e) { /* … */ },
    fail: function (e) { /* … */ }
}));
maskedigits(form.form.phone, '(___) ___-__-__', false);
maskedigits(form.form.birthdate, '__.__.____', false);
pattern_validator.call(form.form, [form.form.fio, form.form.email]); // валидация "на лету" при вводе
typeahead(form.form.city, { url: '/mock/cities', skip: 2 });

        

    


    
    

        

            
2. group() на части полей — не на всю форму

            

                group() необязательно вешать на весь <form> —
                первый аргумент может быть просто списком элементов (так в оригинале собрана строка
                фильтра таблицы: new group(ui.els('[filter-piece]'), fopt)). Здесь
                вторая, полностью независимая от формы выше группа — только на поле уведомлений «по умолчанию»,
                и она же отправляется на сервер отдельно от всей формы.
            

            

                
                

                    
                    
                

                
                
            

            

            
var prefs = new group(ui.els('[data-group="prefs"]'), {
    change: function () { statusEl.textContent = 'изменено: ' + JSON.stringify(prefs.data); }
});
prefs.data; // { digest: 'daily', dark_theme: null } — независимо от формы карточки выше

// эта группа отправляется САМА, отдельным запросом — вся остальная форма
// её вообще не касается:
xhr({ method: 'patch', url: '/mock/users/42/preferences', data: prefs.data, done: ..., fail: ... });


            


            
2b. Отправка только ИЗМЕНИВШИХСЯ полей главной формы

            

                Тот же xhr()/group.data, но с собственным
                несложным diff'ом: запоминаем снимок form.data при
                загрузке (и после каждой удачной отправки), а при сохранении
                сравниваем с текущим состоянием и в xhr() уходят
                только реально поменявшиеся ключи — экономит трафик и не
                перезаписывает на сервере поля, которые пользователь не трогал.
                Готового такого diff'а в jsroll нет — это несложная надстройка
                поверх form.data.
            

            
            
            

            
var snapshot = JSON.stringify(form.data); // точка отсчёта — что было "исходно"

function sendOnlyChanged() {
    var current = form.data, before = JSON.parse(snapshot), changed = {};
    Object.keys(current).forEach(function (k) {
        if (JSON.stringify(current[k]) !== JSON.stringify(before[k])) changed[k] = current[k];
    });
    if (!Object.keys(changed).length) return; // менять нечего
    xhr({ method: 'patch', url: '/mock/users/42', data: changed, done: function () {
        snapshot = JSON.stringify(current); // после удачной отправки — новая точка отсчёта
    }});
}

5. Маскированный ввод — maskedigits(el, pattern, cleared)

            
Принимает только цифры, буквы/спецсимволы игнорируются — маска задаёт, где именно эти цифры окажутся.

            

                

                

                

                

            

            
maskedigits(document.querySelector('#m-phone'), '(___) ___-__-__', false);
maskedigits(document.querySelector('#m-date'),  '__.__.____', false);
maskedigits(document.querySelector('#m-snils'), '___-___-___ __', false);
maskedigits(document.querySelector('#m-card'),  '____ ____ ____ ____', false);
